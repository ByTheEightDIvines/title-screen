cmake_minimum_required(VERSION 3.26)
project(hero_of_legend_title VERSION 1.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (NOT DUSKLIGHT_VERSION)
    set(DUSKLIGHT_VERSION "v2.0.3")
endif ()
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/FetchDusklight.cmake")
add_subdirectory("${DUSKLIGHT_DIR}/sdk" dusklight-sdk EXCLUDE_FROM_ALL)
configure_file(mod.json.in "${CMAKE_CURRENT_BINARY_DIR}/mod.json" @ONLY NEWLINE_STYLE LF)
set(mod_sources src/title_mod.cpp)
if (NOT WIN32)
    if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
        enable_language(OBJCXX)
        set_source_files_properties(src/audio_portable.cpp PROPERTIES LANGUAGE OBJCXX)
    endif ()
    add_library(title_pcm_backend STATIC src/audio_portable.cpp)
    set_target_properties(title_pcm_backend PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(title_pcm_backend PRIVATE vendor/miniaudio)
endif ()
add_mod(hero_of_legend_title FEATURES game SOURCES ${mod_sources}
    MOD_JSON "${CMAKE_CURRENT_BINARY_DIR}/mod.json" RES_DIR res OVERLAY_DIR overlay BUNDLE)
target_include_directories(hero_of_legend_title PRIVATE vendor/miniaudio)
if (WIN32)
    if (MSVC)
        # The pinned SDK uses __cplusplus to decide whether nullptr exists.
        target_compile_options(hero_of_legend_title PRIVATE /Zc:__cplusplus)
    endif ()
    target_link_libraries(hero_of_legend_title PRIVATE winmm)
else ()
    target_link_libraries(hero_of_legend_title PRIVATE title_pcm_backend)
    find_package(Threads REQUIRED)
    target_link_libraries(hero_of_legend_title PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    if (APPLE)
        target_link_libraries(hero_of_legend_title PRIVATE "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
        if (CMAKE_SYSTEM_NAME STREQUAL "iOS")
            target_link_libraries(hero_of_legend_title PRIVATE "-framework Foundation" "-framework AVFoundation")
        endif ()
    elseif (ANDROID)
        target_link_libraries(hero_of_legend_title PRIVATE m log)
    else ()
        target_link_libraries(hero_of_legend_title PRIVATE m)
    endif ()
endif ()
include(CTest)
if (BUILD_TESTING)
    add_executable(test_music_return tools/test_music_return.cpp)
    add_test(NAME five_second_music_return COMMAND test_music_return)
    if (WIN32 AND NOT CMAKE_CROSSCOMPILING)
        add_executable(test_portable_pcm tools/test_portable_pcm.cpp src/audio_portable.cpp)
        target_include_directories(test_portable_pcm PRIVATE vendor/miniaudio)
        target_link_libraries(test_portable_pcm PRIVATE ole32 winmm uuid)
    endif ()
endif ()
